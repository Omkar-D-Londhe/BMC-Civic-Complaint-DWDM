SELECT 
    z.zone_name, 
    COUNT(f.complaint_key) AS total_complaints, 
    ROUND(AVG(f.resolution_days), 2) AS avg_resolution_days
FROM 
    fact_civic_complaints f
JOIN 
    dim_geography g ON f.geo_key = g.geo_key
JOIN 
    dim_zone z ON g.zone_id = z.zone_id
GROUP BY 
    z.zone_name WITH ROLLUP;
